Skip fileExists if path has % (Windows invalid) */}} {{ if not (findRE "%" $assetPath) }} {{ if fileExists $assetPath }} {{ $image = $potentialImage }} {{ $imageResource = resources.Get (printf "images/%s%s" $titleSlug .) }} {{ break }} {{ end }} {{ end }} {{ end }} {{/* For translated pages: inherit image from another translation's front matter only.
